/* ---- Input state ---- */
const Input = { left: false, right: false, up: false, down: false, jump: false,
                jumpPressed: false, pausePressed: false };

function setKey(k, down) {
  if (k === "jump") { if (down && !Input.jump) Input.jumpPressed = true; Input.jump = down; return; }
  if (k === "pause") { if (down) Input.pausePressed = true; return; }
  if (k in Input) Input[k] = down;
}

/* Keyboard */
const KEYMAP = { ArrowLeft: "left", KeyA: "left", ArrowRight: "right", KeyD: "right",
  ArrowUp: "up", KeyW: "up", ArrowDown: "down", KeyS: "down",
  Space: "jump", KeyZ: "jump", KeyP: "pause", Escape: "pause" };
window.addEventListener("keydown", e => {
  const k = KEYMAP[e.code]; if (!k) return;
  e.preventDefault(); if (!e.repeat) setKey(k, true); else if (k !== "jump" && k !== "pause") setKey(k, true);
});
window.addEventListener("keyup", e => { const k = KEYMAP[e.code]; if (k) { e.preventDefault(); setKey(k, false); } });

  const off = e => { e.preventDefault(); btn.classList.remove("on"); setKey(k, false); };
  btn.addEventListener("pointerdown", on);
  btn.addEventListener("pointerup", off);
  btn.addEventListener("pointercancel", off);
  btn.addEventListener("pointerleave", off);
  btn.addEventListener("contextmenu", e => e.preventDefault());
});

function makeSprite(rows, swap) {
  const h = rows.length, w = rows[0].length;
  const c = document.createElement("canvas"); c.width = w; c.height = h;
  const g = c.getContext("2d");
  for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
    let ch = rows[y][x]; if (!ch) continue;
    if (swap && swap[ch]) ch = swap[ch];
    if (PAL[ch]) { g.fillStyle = PAL[ch]; g.fillRect(x, y, 1, 1); }
  }
  return c;
}

function blit(img, x, y, flip) {
  const w = img.width * SCALE, h = img.height * SCALE;
  x = Math.round(x); y = Math.round(y);
  if (flip) { ctx.save(); ctx.translate(x + w, y); ctx.scale(-1, 1);
              ctx.drawImage(img, 0, 0, w, h); ctx.restore(); }
  else ctx.drawImage(img, x, y, w, h);
}
function spin(img, cx, cy, ang) {
  const w = img.width * SCALE, h = img.height * SCALE;
  ctx.save(); ctx.translate(cx, cy); ctx.rotate(ang);
  ctx.drawImage(img, -w / 2, -h / 2, w, h); ctx.restore();
}

/* ---- Particles (sparks, dust, explosions) ---- */
const Particles = {
  list: [],
  burst(x, y, color, n, speed) {
    for (let i = 0; i < (n || 12); i++) {
      const a = Math.random() * Math.PI * 2, s = (speed || 2) * (0.5 + Math.random());
      this.list.push({ x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s - 1,
        life: 30 + Math.random() * 20, color: color || "#ffd400" });
    }
  },
  dust(x, y) { for (let i = 0; i < 4; i++)
    this.list.push({ x: x + Math.random() * 20, y, vx: (Math.random() - 0.5), vy: -Math.random(),
      life: 15, color: "#888" }); },
  update() { this.list = this.list.filter(p => { p.x += p.vx; p.y += p.vy; p.vy += 0.1; return --p.life > 0; }); },
  draw() { this.list.forEach(p => { ctx.globalAlpha = Math.min(1, p.life / 20);
    ctx.fillStyle = p.color; ctx.fillRect(p.x, p.y, 2, 2); }); ctx.globalAlpha = 1; }
};
